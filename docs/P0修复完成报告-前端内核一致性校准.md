### 1️⃣ V2算法API服务层 (`services/v2AlgorithmApi.ts`) - 新建

#### 核心功能
```
✅ 对接后端 HybridScheduler 统一调度引擎

✅ 对接交通管制系统 (TrafficControlSystem)
✅ 完整TypeScript类型定义
```

#### API端点映射
```typescript
// HybridScheduler
GET/PUT  /api/v2/advanced/hybrid-scheduler/config     → 配置管理
POST     /api/v2/advanced/hybrid-scheduler/run         → 混合调度执行
GET      /api/v2/advanced/hybrid-scheduler/fallback-history → 降层历史

// 交通管制
GET      /api/v2/advanced/traffic/zones                → 区域锁状态
GET      /api/v2/advanced/traffic/congestion           → 拥堵热力图数据
GET      /api/v2/advanced/traffic/deadlock/check       → 死锁检测
POST     /api/v2/advanced/traffic/deadlock/resolve     → 死锁解决
GET      /api/v2/advanced/traffic/stats                → 交通统计
POST     /api/v2/advanced/traffic/release-all          → 强制释放所有锁
```

#### 类型定义
```typescript
interface ThetaStarConfig { weight_heuristic, los_check, smooth_factor, corner_penalty }
interface EcbsConfig { omega, max_runtime_ms, conflict_limit, restart_count }
interface SippConfig { safe_interval_margin, max_wait_time }
interface DStarConfig { replan_threshold, lookahead_factor }
interface HybridSchedulerConfig { mode, enable_fallback, fallback_layers, theta_star, ecbs, sipp, d_star }

interface TrafficZoneStatus { zone_id, locked_by, lock_type, waiting_agvs, congestion_level }
interface CongestionHeatmapData { node_id, x, y, congestion_score, agv_count, avg_wait_time }
interface DeadlockCheckResult { has_deadlock, cycle_nodes, involved_agvs, resolution_suggestions }
interface TrafficStats { total_zones, locked_zones, active_agvs, ... }
```

```typescript
// 旧版: 直接调用V1 API
const result = await api.runSchedule();

// 新版: 使用 HybridScheduler + Fallback机制
const result = await v2Api.runHybridSchedule({ force_mode: hybridMode });

  const result = await api.runSchedule();
  message.info('已降级至V1调度引擎');
}
```

#### 核心方法
```typescript
const handleCheckDeadlock = async () => {
  setDeadlockLoading(true);
  try {
    const result = await v2Api.checkDeadlock(); // 调用后端死锁检测API
    setDeadlockResult(result);
    if (result.has_deadlock) {
      message.error(`检测到死锁! 涉及 ${result.involved_agvs.length} 台AGV`);
    } else {
      message.success('系统正常，未检测到死锁');
    }
  } catch (e) {
    message.error('死锁检测失败');
  } finally {
    setDeadlockLoading(false);
  }
};
```
